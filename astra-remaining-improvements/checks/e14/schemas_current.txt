def bounded_json(value,max_bytes=16384):
    def visit(item,depth=0):
        if depth>8:raise ValueError('JSON nesting exceeds 8 levels')
        if isinstance(item,dict):
            if len(item)>128:raise ValueError('JSON object exceeds 128 fields')
            for key,child in item.items():
                if not isinstance(key,str) or not 1<=len(key)<=128:raise ValueError('JSON keys must contain 1 to 128 characters')
                visit(child,depth+1)
        elif isinstance(item,list):
            if len(item)>128:raise ValueError('JSON array exceeds 128 items')
            for child in item:visit(child,depth+1)
        elif item is None or isinstance(item,(str,bool,int)):pass
        elif isinstance(item,float):
            if not math.isfinite(item):raise ValueError('JSON numbers must be finite')
        else:raise ValueError('Only JSON values are supported')
    visit(value)
    if len(json.dumps(value,ensure_ascii=False,allow_nan=False).encode('utf-8'))>max_bytes:raise ValueError(f'JSON value exceeds {max_bytes} bytes')
    return value

class RequestModel(BaseModel):
    model_config=ConfigDict(extra='forbid',allow_inf_nan=False,validate_default=True)
    @field_validator('attributes','labels','params','filters','layout','indexing_options','position','query','visualization',mode='after',check_fields=False)
    @classmethod
    def json_fields(cls,value):return bounded_json(value)

class TokenCreateRequest(RequestModel):
    name:Name
    token_type:Literal['user','ingest','service']='service'
    role:Role='viewer'
    scopes:list[Annotated[str,Field(max_length=40)]]=Field(default_factory=list,max_length=20)
    expires_in_days:Annotated[int,Field(strict=True,ge=1,le=365)]|None=30
    @field_validator('scopes')
    @classmethod
    def valid_scopes(cls,value):
        from app.permissions import ROLE_PERMISSIONS
        known=set().union(*ROLE_PERMISSIONS.values())
        if len(value)!=len(set(value)) or not set(value)<=known:raise ValueError('Scopes must be unique known permissions')
        return value
    @model_validator(mode='after')
    def ingest_scope(self):
        if self.token_type=='ingest' and (self.role!='service_account' or not self.scopes or not set(self.scopes)<={'logs:write','metrics:write','traces:write'}):raise ValueError('Ingest tokens require service_account role and explicit telemetry write scopes only')
        return self

class TimeWindow(RequestModel):
    start_time:Timestamp
    end_time:Timestamp
    @model_validator(mode='after')
    def ordered(self):
        if self.end_time<=self.start_time:raise ValueError('end_time must follow start_time')
        if (self.end_time-self.start_time).total_seconds()>31*86400:raise ValueError('Choose a time range of at most 31 days')
        return self

class NotificationRequest(RequestModel):
    type:Literal['email','webhook','slack','local']='local'
    target:Annotated[str,StringConstraints(strip_whitespace=True,min_length=1,max_length=255)]
    @model_validator(mode='after')
    def target_valid(self):
        from urllib.parse import urlsplit
        if self.type in {'webhook','slack'}:
            url=urlsplit(self.target)
            if url.scheme!='https' or not url.hostname or url.username or url.password or url.fragment or url.port not in (None,443):raise ValueError('Webhook targets require HTTPS without credentials, fragments or custom ports')
        return self

class TraceSpanIn(RequestModel):
    trace_id:Identifier
    span_id:Identifier
    parent_span_id:Identifier|None=None
    service_name:Name
    operation_name:Name
    status:Annotated[str,Field(min_length=1,max_length=32)]='ok'
    start_ts:Timestamp
    end_ts:Timestamp
    dedupe_key:Annotated[str,Field(min_length=1,max_length=128)]|None=None
    attributes:dict[str,Any]=Field(default_factory=dict)
    @model_validator(mode='after')
    def ordered_span(self):
        if self.end_ts<self.start_ts:raise ValueError('Span end must not precede its start')
        if self.parent_span_id==self.span_id:raise ValueError('A span cannot be its own parent')
        return self
